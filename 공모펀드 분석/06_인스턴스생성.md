# 06. 인스턴스 생성 — 실제 값 채우기 (지식 그래프)

> 클래스 정의와 속성 정의를 바탕으로 실제 데이터를 트리플로 변환한다.
> 95,619개 행 → TTL 인스턴스 생성 파이프라인을 설계한다.

---

## 1. 인스턴스 URI 설계

```
:fund/{itm_no}              → 펀드 인스턴스
:company/{or_co_xtn_itt_cd} → 운용회사 인스턴스
:index/{bmrk_nm 해시}        → 벤치마크 인스턴스
:region/{fd_ivst_rgn_desc}  → 투자지역 인스턴스 (7개 고정)
:riskgrade/{zrin_fd_ivst_risk_gcd} → 위험등급 인스턴스 (6개 고정)
```

예시:
```turtle
:fund/KR5174430032
:company/80021
:index/KOSPI200-25-KAP-Bond-01Y-75
:region/국내
:riskgrade/4
```

## 3. 펀드 인스턴스 생성 파이프라인

### 3-1. 클래스 분류 함수

```python
FUND_TYPE_MAP = {
    "주식형":   "EquityFund",
```

### 3-2. 행 → 트리플 변환 함수

```python
from rdflib import Graph, Namespace, Literal, URIRef
from rdflib.namespace import RDF, RDFS, OWL, XSD
import pandas as pd
import hashlib

BASE = Namespace("http://miraeasset.com/ontology/fund#")

def row_to_triples(row: pd.Series, g: Graph):
    ns = BASE
    itm_no = str(row["itm_no"]).strip()
    fund_uri = ns[f"fund/{itm_no}"]

    # 클래스 분류
    fund_type = FUND_TYPE_MAP.get(str(row.get("or_attr_desc", "")), None)
    if fund_type:
        g.add((fund_uri, RDF.type, ns[fund_type]))
    else:
        g.add((fund_uri, RDF.type, ns["PublicFund"]))  # 분류 불가 시 상위 클래스

    # 필수 식별 속성
    g.add((fund_uri, ns["fundId"], Literal(itm_no, datatype=XSD.string)))
    g.add((fund_uri, RDFS.label, Literal(row["itm_nm"], lang="ko")))
    g.add((fund_uri, ns["fundName"], Literal(row["itm_nm"], datatype=XSD.string)))

    if pd.notna(row.get("itm_abrv_nm")):
        g.add((fund_uri, ns["fundShortName"], Literal(row["itm_abrv_nm"], datatype=XSD.string)))

    # 운용회사 (ObjectProperty)
    if pd.notna(row.get("or_co_xtn_itt_cd")):
        company_code = str(int(row["or_co_xtn_itt_cd"]))
        company_uri = ns[f"company/{company_code}"]
        g.add((fund_uri, ns["managedBy"], company_uri))

    # 위험등급 (ObjectProperty)
    risk_raw = str(row.get("zrin_fd_ivst_risk_grd_nm", "")).strip()
    risk_uri_key = RISK_GRADE_MAP.get(risk_raw)
    if risk_uri_key:
        g.add((fund_uri, ns["hasRiskGrade"], ns[risk_uri_key]))

    # 투자지역 (ObjectProperty)
    region_raw = str(row.get("fd_ivst_rgn_desc", "")).strip()
    region_uri_key = REGION_MAP.get(region_raw)
    if region_uri_key:
        g.add((fund_uri, ns["investsIn"], ns[region_uri_key]))

    # 벤치마크 (ObjectProperty)
    if pd.notna(row.get("bmrk_nm")):
        idx_hash = hashlib.md5(str(row["bmrk_nm"]).encode()).hexdigest()[:8]
        idx_uri = ns[f"index/{idx_hash}"]
        g.add((fund_uri, ns["followsBenchmark"], idx_uri))
        g.add((idx_uri, RDF.type, ns["BenchmarkIndex"]))
        g.add((idx_uri, RDFS.label, Literal(row["bmrk_nm"], lang="ko")))
        if pd.notna(row.get("bmrk_eng_nm")):
            g.add((idx_uri, RDFS.label, Literal(row["bmrk_eng_nm"], lang="en")))

    # 수익률 (DatatypeProperty, 선택적)
    return_cols = {
        "fd_wk1_ern_r":  "returnWeek1",
        "fd_mm1_ern_r":  "returnMonth1",
        "fd_mm3_ern_r":  "returnMonth3",
        "fd_mm6_ern_r":  "returnMonth6",
        "fd_yr1_ern_r":  "returnYear1",
        "fd_yr2_ern_r":  "returnYear2",
        "fd_yr3_ern_r":  "returnYear3",
        "fd_yr5_ern_r":  "returnYear5",
        "fd_mm18_ern_r": "returnMonth18",
    }
    for col, prop in return_cols.items():
        val = row.get(col)
        if pd.notna(val):
            g.add((fund_uri, ns[prop], Literal(float(val), datatype=XSD.decimal)))

    # 순자산
    if pd.notna(row.get("fd_nast_suma")) and row["fd_nast_suma"] >= 0:
        g.add((fund_uri, ns["netAsset"], Literal(float(row["fd_nast_suma"]), datatype=XSD.decimal)))

    # 판매 상태
    if pd.notna(row.get("sale_yn")):
        g.add((fund_uri, ns["saleStatus"], Literal(str(row["sale_yn"]), datatype=XSD.string)))

    # 환헤지
    exchdg = str(row.get("exchdg_yn", "")).strip()
    if exchdg == "Y":
        g.add((fund_uri, ns["isHedged"], Literal(True, datatype=XSD.boolean)))
    elif exchdg == "N":
        g.add((fund_uri, ns["isHedged"], Literal(False, datatype=XSD.boolean)))
    # "00080008" → 생략 (NULL)

    # 기타 boolean
    bool_cols = {
        "hdge_fd_yn": "isHedgeFund",
        "ofsfd_yn":   "isForeignFund",
        "frc_bpr_itm_yn": "isForeignCurrencyBase",
    }
    for col, prop in bool_cols.items():
        val = row.get(col)
        if pd.notna(val):
            g.add((fund_uri, ns[prop], Literal(bool(int(val)), datatype=XSD.boolean)))

    # 당사 판매 여부
    thco = str(row.get("thco_sale_yn", "")).strip()
    if thco == "Y":
        g.add((fund_uri, ns["miPoSaleStatus"], Literal(True, datatype=XSD.boolean)))
    elif thco == "N":
        g.add((fund_uri, ns["miPoSaleStatus"], Literal(False, datatype=XSD.boolean)))

    return g
```

### 3-3. 전체 실행 루프

```python
def build_fund_graph(df: pd.DataFrame) -> Graph:
    g = Graph()
    g.bind("", BASE)
    g.bind("owl", OWL)
    g.bind("rdfs", RDFS)
    g.bind("xsd", XSD)

    skipped = 0
    for _, row in df.iterrows():
        # 필수 속성 검증
        if pd.isna(row.get("itm_no")) or pd.isna(row.get("itm_nm")):
            skipped += 1
            continue
        if pd.isna(row.get("or_co_xtn_itt_cd")):
            skipped += 1
            continue
        row_to_triples(row, g)

    print(f"처리 완료: {len(df) - skipped}건, 스킵: {skipped}건")
    print(f"총 트리플 수: {len(g)}")
    return g

# 실행
df = pd.read_excel("PRFD01N001_공모펀드마스터_20260711_datarows.xlsx")
g = build_fund_graph(df)
g.serialize("fund_ontology.ttl", format="turtle")
```

## 6. 인스턴스 생성 시 주의사항

| 이슈 | 처리 방법 |
|------|----------|
| 같은 벤치마크가 다른 표기로 등장 | 전처리 단계에서 벤치마크명 정규화 후 해시 생성 |
| 운용회사 코드가 float (예: 40010.0) | `int()` 변환 후 문자열화 |
| 수익률이 object 타입으로 읽힘 | `pd.to_numeric(errors='coerce')` 적용 |
| 판매완료 펀드도 포함 여부 | 포함 (질문에서 "과거 판매 펀드"도 조회 가능해야 함) |
